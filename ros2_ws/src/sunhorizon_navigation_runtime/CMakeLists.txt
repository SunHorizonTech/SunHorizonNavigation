project(sunhorizon_navigation_runtime)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)

endforeach()
add_library(navigation_runtime src/navigation_runtime.cpp src/nav2_assembly.cpp)
target_include_directories(navigation_runtime PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(navigation_runtime ${dependencies})
target_link_libraries(navigation_runtime
  sunhorizon_obstacle_coverage::coverage_runtime
  sunhorizon_camera_runtime::camera_runtime
  sunhorizon_airy_adapter::airy_adapter_runtime
  sunhorizon_rgbd_adapter::pointcloud_self_exclusion_runtime
  sunhorizon_perception_runtime::perception_runtime
  sunhorizon_livo_bringup::livo_direct_adapter
  sunhorizon_3d_localizer::sunhorizon_ndt_direct_runtime
  sunhorizon_localization_manager::localization_manager_runtime
  fast_livo::laser_mapping rslidar_sdk::rslidar_direct_driver Threads::Threads)
target_compile_options(navigation_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sunhorizon_navigation_runtime_node src/main.cpp)
target_link_libraries(sunhorizon_navigation_runtime_node navigation_runtime)
ament_target_dependencies(sunhorizon_navigation_runtime_node ${dependencies})
target_compile_options(sunhorizon_navigation_runtime_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(diagnostic_identity_observer src/diagnostic_identity_observer.cpp)
ament_target_dependencies(diagnostic_identity_observer rclcpp diagnostic_msgs)
target_compile_options(diagnostic_identity_observer PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS navigation_runtime EXPORT export_navigation_runtime
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS sunhorizon_navigation_runtime_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS diagnostic_identity_observer RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_navigation_runtime HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies} Threads)
