cmake_minimum_required(VERSION 3.16)
project(sunhorizon_map_annotation_rviz)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(map_annotation_panel SHARED src/panel.cpp)
ament_target_dependencies(map_annotation_panel rviz_common pluginlib rclcpp geometry_msgs visualization_msgs)
target_link_libraries(map_annotation_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS map_annotation_panel LIBRARY DESTINATION lib)
ament_package()
